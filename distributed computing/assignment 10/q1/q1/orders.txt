κανω μια αλλαγη στον client γιατι θελουμε παντα να ρωταει τον χρηστη για το n
αλλαζω το:
int iters = 10000;
		for (int i = 0; i < iters; i++)  {

			out.println(outmsg);
			inmsg = in.readLine();
			app.processReply(inmsg);
		}

		out.println(outmsg);

outmsg = app.prepareRequest();
        while(!outmsg.equals(EXIT)) {
